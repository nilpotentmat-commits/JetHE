cmake_minimum_required(VERSION 3.16)
project(jethe_helib_controls LANGUAGES CXX)
find_package(helib REQUIRED)
add_executable(helib_slack_control src/leveled/slack_profile.cpp)
add_executable(helib_core_matched src/core/core_measure.cpp)
foreach(target helib_slack_control helib_core_matched)
  target_compile_features(${target} PRIVATE cxx_std_17)
  target_compile_options(${target} PRIVATE -Wall -Wextra)
  target_include_directories(${target} PRIVATE vendor)
  target_link_libraries(${target} PRIVATE helib)
endforeach()
